```python
class TestHTTPDurationHistogram:
    def test_post_request_records_histogram_observation(self, client):
        before = _histogram_count(
            "aria_http_request_duration_seconds",
            {"method": "POST", "status_code": "200"},
        )

        # Use any non-skipped route that returns 200 (e.g. /ingest/text with valid body)

        assert resp.status_code == 200

        after = _histogram_count(
            "aria_http_request_duration_seconds",
            {"method": "POST", "status_code": "200"},
        )
        assert after == before + 1

    def test_skip_paths_do_not_record_histogram(self, client):
        before = _histogram_count(
            "aria_http_request_duration_seconds",
            {"method": "GET", "status_code": "200"},
        )

        client.get("/health")

        after = _histogram_count(
            "aria_http_request_duration_seconds",
            {"method": "GET", "status_code": "200"},
        )
        assert after == before  # /health is in _SKIP_PATHS; no observation
```

```python
async def test_sqlite_write_failure_on_success_increments_error_counter_and_warns(
    telemetry_store: TelemetryStore, caplog
) -> None:
    import logging
    structlog.contextvars.clear_contextvars()

    before = TELEMETRY_WRITE_ERRORS_COUNTER.labels(source="llm")._value.get()

    mock_response = _mock_success_response(response_cost=0.001)

    with (
        patch("litellm.acompletion", new_callable=AsyncMock, return_value=mock_response),
        patch.object(
            telemetry_store,
            "record_llm_call",
            side_effect=sqlite3.OperationalError("disk full"),
        ),
        caplog.at_level(logging.WARNING, logger="aria.llm.client"),
    ):
        from aria.llm.client import LLMClient
        llm = LLMClient()
        result = await llm.complete([{"role": "user", "content": "hi"}])

    assert result == "ok"  # call still succeeded
    after = TELEMETRY_WRITE_ERRORS_COUNTER.labels(source="llm")._value.get()
    assert after == before + 1
    assert any("telemetry store write failed" in r.message for r in caplog.records)
    assert any("OperationalError" in r.message for r in caplog.records)
```

```python
class TestCostByRequest:
    def test_returns_sum_of_nonull_cost_rows(self, telemetry_store: TelemetryStore) -> None:
        from datetime import datetime, timezone
        ts = datetime.now(timezone.utc)
        telemetry_store.record_llm_call(
            request_id="req-A", model="gpt-4o", latency_ms=100.0,
            status="success", attempt=1, cost_usd=0.001, ts=ts,
        )
        telemetry_store.record_llm_call(
            request_id="req-A", model="gpt-4o", latency_ms=80.0,
            status="success", attempt=1, cost_usd=0.002, ts=ts,
        )

        result = telemetry_store.cost_by_request("req-A")
        assert result == pytest.approx(0.003)

    def test_returns_none_when_cost_usd_is_null(self, telemetry_store: TelemetryStore) -> None:
        from datetime import datetime, timezone
        ts = datetime.now(timezone.utc)
        telemetry_store.record_llm_call(
            request_id="req-B", model="ollama/llama3.2", latency_ms=200.0,
            status="success", attempt=1, cost_usd=None, ts=ts,
        )

        result = telemetry_store.cost_by_request("req-B")
        assert result is None

    def test_returns_none_for_unknown_request_id(self, telemetry_store: TelemetryStore) -> None:
        result = telemetry_store.cost_by_request("no-such-request-id")
        assert result is None

    def test_sums_multiple_rows_for_same_request_id(self, telemetry_store: TelemetryStore) -> None:
        """Repair path in complete_structured produces two rows per request_id."""
        from datetime import datetime, timezone
        ts = datetime.now(timezone.utc)
        for _ in range(3):
            telemetry_store.record_llm_call(
                request_id="req-C", model="gpt-4o", latency_ms=50.0,
                status="success", attempt=1, cost_usd=0.0005, ts=ts,
            )
        result = telemetry_store.cost_by_request("req-C")
        assert result == pytest.approx(0.0015)
```
